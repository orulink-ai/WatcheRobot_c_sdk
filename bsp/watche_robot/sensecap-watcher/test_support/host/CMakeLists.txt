function(extract_function path signature output)
    file(READ "${path}" source)
    string(REPLACE "\r\n" "\n" source "${source}")
    string(FIND "${source}" "${signature}" begin)
    if(begin LESS 0)
        message(FATAL_ERROR "Missing ${signature}")
    endif()
    string(SUBSTRING "${source}" ${begin} -1 tail)
    string(FIND "${tail}" "\n}" end)
    math(EXPR length "${end}+3")
    string(SUBSTRING "${tail}" 0 ${length} body)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${body}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
endfunction()
