cmake_minimum_required(VERSION 3.20)
project(inference_startup_tests C)
enable_testing()
set(source_path "${CMAKE_CURRENT_LIST_DIR}/../../src/hal_camera_ptl.c")
file(READ "${source_path}" source)
string(REPLACE "\r\n" "\n" source "${source}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_path}")
string(FIND "${source}" "typedef struct {" begin)
string(FIND "${source}" "} hal_camera_ptl_context_t;" end)
string(LENGTH "} hal_camera_ptl_context_t;" suffix_length)
math(EXPR length "${end}-${begin}+${suffix_length}")
string(SUBSTRING "${source}" ${begin} ${length} context)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/context.inc" "${context}\n")
string(FIND "${source}" "static void inference_task(" begin)
string(SUBSTRING "${source}" ${begin} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end}+3")
string(SUBSTRING "${tail}" 0 ${length} worker)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/inference_task.inc" "${worker}\n")
add_executable(inference_startup test_inference_startup.c)
set_property(SOURCE test_inference_startup.c APPEND PROPERTY OBJECT_DEPENDS
    "${CMAKE_CURRENT_BINARY_DIR}/context.inc;${CMAKE_CURRENT_BINARY_DIR}/inference_task.inc")
target_include_directories(inference_startup PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../../include
    ../../../watcher_hx_ptl/include ../../../watcher_hx_ptl/test_support/host/stubs)
foreach(scenario initial after_frame timeout boundary recovery wrong_generation error)
    add_test(NAME inference_startup_${scenario} COMMAND inference_startup ${scenario})
endforeach()
